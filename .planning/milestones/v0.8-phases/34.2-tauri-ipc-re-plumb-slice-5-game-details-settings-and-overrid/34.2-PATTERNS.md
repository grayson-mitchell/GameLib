export function setGameOverrides(appName: string, override: GameMetadataOverride): boolean {
  try {

// D-11/D-16: three-state map — key absent (never looked up) vs null
// (looked up, no record) vs number (rating). Preserve this contract exactly.
export async function buildCrossoverRatingMap(): Promise<Record<string, number | null>> {
  const map: Record<string, number | null> = {}
  if (!isMac) {
    return map
  }
  for (const manager of Object.values(libraryManagerMap)) {
    for (const gameInfo of manager.getListOfGames()) {
      if (!isCrossoverIndexEligible(gameInfo)) {
        continue // Never looked up — key stays absent (D-16).
      }
      const result = await getCodeweaversFromIndex(gameInfo)
      map[gameInfo.app_name] = result?.macRating ?? null
    }
  }
  return map
}
