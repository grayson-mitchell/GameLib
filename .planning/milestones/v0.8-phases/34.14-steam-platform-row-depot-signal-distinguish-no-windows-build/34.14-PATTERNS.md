```typescript
export function selectSteamPlatformOptions<T extends { value: InstallPlatform }>(
  mode: SteamPlatformRowMode,
  platforms: T[],
  hasWindowsDepot: boolean
): T[] {
  switch (mode) {
    case 'absent':
      return []
    case 'readonly-windows': {
      const windowsEntry = platforms.find((p) => p.value === 'Windows')
      return windowsEntry ? [windowsEntry] : []
    }
    case 'readonly-macos': {
      const macEntry = platforms.find((p) => p.value === 'Mac')
      return macEntry ? [macEntry] : []
    }
    case 'selectable': {
      const macEntry = platforms.find((p) => p.value === 'Mac')
      const windowsEntry = hasWindowsDepot
        ? platforms.find((p) => p.value === 'Windows')
        : undefined
      return [macEntry, windowsEntry].filter((p): p is T => Boolean(p))
    }
  }
}

    const alreadyCaptured = (): boolean =>
      steamMetadataStore.get(this.appId)?.platformsCaptured === true

    if (alreadyCaptured()) return

    await this.fetchMetadataIfNeeded(this.getGameInfo())

    if (!alreadyCaptured() && pendingFetches.has(this.appId)) {
      const deadline = Date.now() + METADATA_FETCH_TIMEOUT_MS
      while (
        !alreadyCaptured() &&
        pendingFetches.has(this.appId) &&
        Date.now() < deadline
      ) {
        await new Promise((resolve) => setTimeout(resolve, 100))
      }
    }
  }
```

## Shared Patterns

### Pattern: additive widening of an already-round-tripped object

**Source:** `SteamBottleEligibilityVerdict` (`common/types/steam.ts:59-63`)

### Pattern: pure-function extraction, zero runtime import of `index.tsx`

**Source:** every module header in `steamSectionGating.ts` (lines 13-24) and
`steamPlatformRow.ts` (lines 10-20)
**Apply to:** ANY new logic this phase adds — must live in `steamSectionGating.ts`,
`steamPlatformRow.ts`, `steamEligibilityProbe.ts`, or a new sibling module. `index.tsx`'s first
line is `import './index.scss'`; the Frontend jest project has no jsdom/CSS transform
(confirmed via `src/frontend/jest.config.js`, cited but not independently re-read this session —
RESEARCH.md's citation is treated as reliable given it is a config file, not a moving target).

### Pattern: post-processing composition instead of inline conditionals

**Source:** `index.tsx:349-352` — `applyLibraryFetchPending(applyEligibilityPending(raw,
eligibility.pending), steamLibrariesSettled)`
**Apply to:** D-04's fail-open override. RESEARCH.md's OQ5/(b) analysis recommends a THIRD
composed function following this exact chaining shape, not a fourth inline term in
`resolveSteamSectionGating`'s own priority chain (which cannot see `eligibility.pending` at
all — it is a pure function one layer below `index.tsx`).

### Pattern: explicit skip-list over negated comparison

**Source:** `shouldProbeEligibility` (`steamEligibilityProbe.ts:65-72`),
`resolveSteamSectionGating` step 1 (`steamSectionGating.ts:143-150`)
**Apply to:** any new host-platform or state check this phase introduces — write `!== 'X' &&
!== 'Y'`, never `=== onlyExpectedValue`, so an unrecognised value still asks the backend /
fails toward the safer branch.

### Pattern: read-order enforcement as the "single seam" for a meaningless-combination obligation

**Source:** `steamSectionGating.ts` step 2, `effectiveBottleRequired = isMac &&
input.bottleRequired` (lines 153-158)
**Apply to:** D-03's `(hasWindowsDepot: true, platformsCaptured: false)` obligation — enforce by
placing the `!platformsCaptured` check structurally BEFORE the `hasWindowsDepot` check inside
`resolveSteamSectionGating`'s existing if/else chain (RESEARCH.md OQ5's recommendation), backed
by a defensive-normalisation test (see below), not a discriminated-union type.
