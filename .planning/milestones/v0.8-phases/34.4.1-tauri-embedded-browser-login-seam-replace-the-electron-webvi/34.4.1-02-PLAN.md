  export interface LoginWindowCookie { name: string; domain: string | null; value: string }
  export interface LoginWindowCookieRead { total: number; matched: LoginWindowCookie[] }
  export interface LoginWindowNavEvent { event: 'started' | 'finished'; url: string }

    open(url: string, options: { visible: boolean; userAgent: string }): Promise<string>
    cookies(label: string, host: string, names: string[]): Promise<LoginWindowCookieRead>
    takeEvents(label: string): Promise<LoginWindowNavEvent[]>
    close(label: string): Promise<boolean>
    clearCookies(label: string, domain: string): Promise<number>

    | 'SUPPORTED_NONEMPTY' | 'SUPPORTED_BUT_EMPTY' | 'UNSUPPORTED_OR_ERROR' | 'UNDECIDABLE'

    total: number | null      // null == the read threw
