export interface TimerOptions { keepAlive?: { reason: string } }

  clear(handle: TimerHandle): void

export interface ChildHandle { readonly pid: number | undefined; kill(signal?: NodeJS.Signals): boolean }
export interface TrackFilter { runner?: string; key?: string }
export interface RunOptions {
  cwd?: string
  env?: NodeJS.ProcessEnv
  replaceEnv?: boolean            // true: env is used verbatim (runWineCommand shape); default merges over baseEnv
  timeoutMs?: number              // default DEFAULT_RUN_TIMEOUT_MS; must be positive finite
  signal?: AbortSignal
  stdin?: string
  onSpawn?: (child: ChildHandle) => void
  onStdout?: (chunk: string, child: ChildHandle) => void
  onStderr?: (chunk: string, child: ChildHandle) => void

  sanitize?: (text: string) => string
  track?: { runner: string; key: string }

  maxBufferBytes?: number         // default 8 MiB per stream; capture truncates, sets truncated
  windowsHide?: boolean           // default true
}
export interface RunResult {
  code: number | null; signal: NodeJS.Signals | null; stdout: string; stderr: string
  timedOut: boolean; aborted: boolean; truncated: boolean; pid: number | undefined
}
export interface LongLivedOptions extends Omit<RunOptions, 'timeoutMs' | 'resolveOn' | 'maxBufferBytes' | 'stdin'> {
  id: string
  stdio?: 'pipe' | 'ignore'
  teardown?: (handle: LongLivedHandle) => void   // default: handle.kill('SIGTERM')
}

}

  openDetached(url: string, onError?: (code: string) => void): void
