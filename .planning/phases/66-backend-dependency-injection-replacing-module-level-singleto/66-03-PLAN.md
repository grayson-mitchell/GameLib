Contract this plan creates (66-08 and later phases use these names):

```typescript
export interface ScriptStep {
  cmd: string
  argv?: readonly string[]               // exact match when given; omitted = any argv
  match?: (argv: readonly string[]) => boolean
  result?: Partial<RunResult>            // defaults: code 0, stdout '', stderr '', flags false
  chunks?: { stream: 'stdout' | 'stderr'; text: string }[]   // delivered to onStdout/onStderr before resolve
  error?: { code: string }               // run() rejects with this errno code (e.g. ENOENT)
  repeat?: boolean                       // step stays matchable after use
}
export interface ScriptedCall { method: 'run' | 'runSync' | 'spawnLongLived' | 'openDetached'; cmd: string; argv: readonly string[]; env?: NodeJS.ProcessEnv; cwd?: string; opts?: unknown }
export interface ScriptedRunner extends ProcessRunner { readonly calls: readonly ScriptedCall[]; expectAllConsumed(): void }
export function createScriptedRunner(script: readonly ScriptStep[]): ScriptedRunner

export interface FakeClock extends Clock { advance(ms: number): void; runAll(): void; pendingCount(): number; keepAliveReasons(): string[]; assertNoKeepAlive(): void }
export function createFakeClock(startMs?: number): FakeClock

export interface CaptureLog extends LogSink { readonly lines: readonly { level: 'info' | 'warn' | 'error'; msg: string }[] }
export function createCaptureLog(): CaptureLog

export function createRealRunner(opts?: { realProfile?: { justification: string }; clock?: Clock; log?: LogSink }): { runner: ProcessRunner; profile: FakeHomeProfile | null; dispose(): void }

export interface TestContext { ctx: AppContext; runner: ScriptedRunner; clock: FakeClock; log: CaptureLog; profile: FakeHomeProfile; dispose(): void }
export function createTestContext(overrides?: { script?: readonly ScriptStep[]; platform?: NodeJS.Platform; paths?: Partial<AppPaths>; runner?: ProcessRunner }): TestContext
```
