- REST (owner/repo from args, base https://api.github.com):
  1 GET /repos/{o}/{r}/contents/{path}?ref={ref} -> 200 {sha, content, encoding:'base64'} | 404
  2 GET /repos/{o}/{r}/pulls?head={o}:{branch}&state={open|closed}&per_page=10 -> [{number, state, merged_at, body}]
  3 GET /repos/{o}/{r}/git/ref/heads/{branch} -> {object:{sha}}
  4 POST /repos/{o}/{r}/git/refs {ref:'refs/heads/{branch}', sha} -> 201 | 422 'Reference already exists'
  5 PATCH /repos/{o}/{r}/git/refs/heads/{branch} {sha, force:true}
  6 PUT /repos/{o}/{r}/contents/{path} {message, content(base64), branch, sha?} -> 200/201 {content:{sha}}
  7 DELETE /repos/{o}/{r}/contents/{path} {message, sha, branch}
  8 POST /repos/{o}/{r}/pulls {title, head, base, body} -> 201 {number}
  9 PATCH /repos/{o}/{r}/pulls/{n} {body}
  10 GET /repos/{o}/{r}/issues?labels={label}&state=all&per_page=100 -> [{number, title, pull_request?}]
  11 POST /repos/{o}/{r}/issues {title, body, labels} -> 201 {number}
</interfaces>
